## 侧边栏自动生成

### sidebar.ts

```ts
// docs/.vitepress/sidebar.ts
import fs from 'node:fs'
import path from 'node:path'

/**
 * 扫描目录，自动生成侧边栏
 * 文件名格式: 01.xxx.md → 按数字前缀排序
 */
function autoSidebar(
  dir: string,
  title: string,
  options: {
    /** 图标前缀 */
    icon?: string
    /** 是否默认折叠 */
    collapsed?: boolean
    /** 是否隐藏 index.md */
    hideIndex?: boolean
  } = {}
) {
  const { icon = '', collapsed = false, hideIndex = true } = options
  const dirPath = path.resolve(__dirname, '..', dir)

  if (!fs.existsSync(dirPath)) return []

  const files = fs.readdirSync(dirPath)
    .filter(f => f.endsWith('.md'))
    .filter(f => (hideIndex ? f !== 'index.md' : true))
    .sort((a, b) => {
      const numA = parseInt(a.match(/^(\d+)/)?.[1] ?? '999')
      const numB = parseInt(b.match(/^(\d+)/)?.[1] ?? '999')
      if (numA !== numB) return numA - numB
      return a.localeCompare(b)
    })

  const items = files.map(f => {
    // 从文件名提取标题：去掉数字前缀和 .md 后缀
    const name = f.replace(/^\d+\./, '').replace(/\.md$/, '')
    const link = `/${dir}/${f.replace(/\.md$/, '')}`
    return { text: name, link }
  })

  if (!items.length) return []

  return [{
    text: icon ? `${icon} ${title}` : title,
    items,
    collapsed,
  }]
}

/**
 * 扫描子目录，为每个子目录生成独立的侧边栏分组
 */
function autoSidebarWithSubDirs(
  dir: string,
  title: string,
  options: { icon?: string; collapsed?: boolean } = {}
) {
  const { icon = '', collapsed = false } = options
  const dirPath = path.resolve(__dirname, '..', dir)

  if (!fs.existsSync(dirPath)) return []

  const groups: Array<{ text: string; items: Array<{ text: string; link: string }>; collapsed: boolean }> = []

  // 读取子目录
  const subDirs = fs.readdirSync(dirPath, { withFileTypes: true })
    .filter(d => d.isDirectory())
    .sort((a, b) => a.name.localeCompare(b.name))

  for (const subDir of subDirs) {
    const subDirPath = path.join(dirPath, subDir.name)
    const files = fs.readdirSync(subDirPath)
      .filter(f => f.endsWith('.md') && f !== 'index.md')
      .sort((a, b) => {
        const numA = parseInt(a.match(/^(\d+)/)?.[1] ?? '999')
        const numB = parseInt(b.match(/^(\d+)/)?.[1] ?? '999')
        if (numA !== numB) return numA - numB
        return a.localeCompare(b)
      })

    if (!files.length) continue

    // 子目录名作为分组标题（去掉数字前缀）
    const groupTitle = subDir.name.replace(/^\d+\./, '')

    groups.push({
      text: groupTitle,
      items: files.map(f => ({
        text: f.replace(/^\d+\./, '').replace(/\.md$/, ''),
        link: `/${dir}/${subDir.name}/${f.replace(/\.md$/, '')}`
      })),
      collapsed,
    })
  }

  // 根目录下的 md 文件（不在子目录中）
  const rootFiles = fs.readdirSync(dirPath)
    .filter(f => f.endsWith('.md') && f !== 'index.md')
    .sort((a, b) => {
      const numA = parseInt(a.match(/^(\d+)/)?.[1] ?? '999')
      const numB = parseInt(b.match(/^(\d+)/)?.[1] ?? '999')
      if (numA !== numB) return numA - numB
      return a.localeCompare(b)
    })

  if (rootFiles.length) {
    groups.unshift({
      text: '概览',
      items: rootFiles.map(f => ({
        text: f.replace(/^\d+\./, '').replace(/\.md$/, ''),
        link: `/${dir}/${f.replace(/\.md$/, '')}`
      })),
      collapsed: true,
    })
  }

  if (!groups.length) return []

  return [{
    text: icon ? `${icon} ${title}` : title,
    items: groups.flatMap(g => g.items),  // sugar-blog 侧边栏格式
  }]
}

// ====== 导出侧边栏配置 ======

// 指南区：平铺结构
const guideSidebar = autoSidebar('guide', '指南', { icon: '📖' })

// 博客区：按子目录分组
const postsSidebar = autoSidebarWithSubDirs('posts', '文章', { icon: '✍️' })

export const sidebar = {
  '/guide/': guideSidebar,
  '/posts/': postsSidebar,
  // 归档和标签页不需要侧边栏
  '/archive/': [],
  '/tags/': [],
}
```

## 归档与标签页

### 归档数据加载器

```ts
// docs/archive/archive.data.ts
import { createContentLoader } from 'vitepress'

interface ArchiveItem {
  title: string
  url: string
  date: string
  year: number
}

interface ArchiveByYear {
  year: number
  count: number
  items: ArchiveItem[]
}

export default createContentLoader('posts/**/*.md', {
  transform(rawData): ArchiveByYear[] {
    const items: ArchiveItem[] = rawData
      .filter(item => !item.frontmatter.hidden && item.frontmatter.date)
      .sort((a, b) => +new Date(b.frontmatter.date) - +new Date(a.frontmatter.date))
      .map(item => {
        const d = new Date(item.frontmatter.date)
        return {
          title: item.frontmatter.title || '无标题',
          url: item.url,
          date: `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`,
          year: d.getFullYear(),
        }
      })

    const grouped = new Map<number, ArchiveItem[]>()
    for (const item of items) {
      if (!grouped.has(item.year)) grouped.set(item.year, [])
      grouped.get(item.year)!.push(item)
    }

    return Array.from(grouped.entries())
      .map(([year, items]) => ({ year, count: items.length, items }))
      .sort((a, b) => b.year - a.year)
  }
})
```

### 标签数据加载器

```ts
// docs/tags/tags.data.ts
import { createContentLoader } from 'vitepress'

interface TagInfo {
  name: string
  count: number
  posts: { title: string; url: string; date: string }[]
}

export default createContentLoader('posts/**/*.md', {
  transform(rawData): TagInfo[] {
    const tagMap = new Map<string, TagInfo>()

    for (const item of rawData) {
      if (item.frontmatter.hidden) continue

      const tags: string[] = item.frontmatter.tags || item.frontmatter.tag || []
      const date = item.frontmatter.date
        ? new Date(item.frontmatter.date).toISOString().slice(0, 10)
        : ''

      for (const tag of tags) {
        if (!tagMap.has(tag)) {
          tagMap.set(tag, { name: tag, count: 0, posts: [] })
        }
        const info = tagMap.get(tag)!
        info.count++
        info.posts.push({
          title: item.frontmatter.title || '无标题',
          url: item.url,
          date,
        })
      }
    }

    return Array.from(tagMap.values())
      .sort((a, b) => b.count - a.count)
      .map(tag => ({
        ...tag,
        posts: tag.posts.sort((a, b) => b.date.localeCompare(a.date))
      }))
  }
})
```

## 部署上线

### GitHub Actions 自动部署

创建 `.github/workflows/deploy.yml`：

```yaml
name: Deploy Blog

on:
  push:
    branches: [main]

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages
  cancel-in-progress: true

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - uses: pnpm/action-setup@v4
        with:
          version: 9

      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: pnpm

      - name: Install dependencies
        run: pnpm install

      - name: Build
        run: pnpm build

      - name: Upload artifact
        uses: actions/upload-pages-artifact@v3
        with:
          path: docs/.vitepress/dist

  deploy:
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - name: Deploy to GitHub Pages
        id: deployment
        uses: actions/deploy-pages@v4
```

### 部署步骤

```bash
# 1. 创建 GitHub 仓库

# 2. 推送代码
git init
git add .
git commit -m "init: my tech blog"
git remote add origin https://github.com/your-username/blog.git
git push -u origin main

# 3. 仓库 Settings → Pages → Source → GitHub Actions

# 4. 推送后自动部署，访问 https://your-username.github.io/blog/
```
