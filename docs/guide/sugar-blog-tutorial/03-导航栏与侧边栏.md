# ③ 导航栏与侧边栏

导航栏和侧边栏在 `docs/.vitepress/config.mts` 中配置。

## 基础配置结构

```typescript
export default defineConfig({
  extends: blogTheme,  // 继承 sugar-blog 主题
  base,                 // 基础路径（部署相关）
  lang: 'zh-cn',
  title: '我的博客',
  description: '博客描述',
  
  themeConfig: {
    nav: [ /* 导航栏配置 */ ],
    sidebar: { /* 侧边栏配置 */ },
    socialLinks: [ /* 社交链接 */ ],
  }
})
```

## 导航栏配置

导航栏支持一级菜单和下拉菜单：

```typescript
nav: [
  // 一级菜单（直接跳转）
  { text: '首页', link: '/' },
  
  // 下拉菜单
  {
    text: '教程',
    items: [
      {
        text: 'Vitepress教程',
        items: [
          { text: '📖 概览', link: '/guide/vitepress-tutorial/' },
          { text: '① 首页 + 导航栏', link: '/guide/vitepress-tutorial/① 首页 + 导航栏' },
        ]
      },
      {
        text: '博客搭建',
        items: [
          { text: '📖 概览', link: '/guide/sugar-blog-tutorial/' },
        ]
      }
    ]
  },
  
  // 一级菜单
  { text: '关于', link: '/about' },
  
  // 带版本号的下拉菜单
  {
    text: '版本',
    items: [
      { text: '当前版本', items: [
        { text: 'v2.0.0 (最新)', link: '/versions/v2' },
      ]},
      { text: '变更日志', link: '/changelog' }
    ]
  },
  
  { text: '归档', link: '/archives' },
]
```

### 导航 `link` 的路径规则

- `/` — 首页
- `/about` — 对应 `docs/about.md`
- `/guide/xxx` — 对应 `docs/guide/xxx.md`
- `/posts/xxx` — 对应 `docs/posts/xxx.md`

::: warning 注意
路径不需要写 `.md` 后缀，VitePress 会自动处理。
:::

## 侧边栏配置

侧边栏按路径分组配置：

```typescript
sidebar: {
  // /guide/ 路径下的页面使用这个侧边栏
  '/guide/': [
    {
      text: 'Vitepress教程', 
      items: [
        { text: '📖 概览', link: '/guide/vitepress-tutorial/' },
        { text: '① 首页 + 导航栏', link: '/guide/vitepress-tutorial/① 首页 + 导航栏' },
        // ... 更多文章
      ]
    },
    {
      text: '博客搭建教程',
      items: [
        { text: '📖 概览', link: '/guide/sugar-blog-tutorial/' },
        { text: '① 项目初始化', link: '/guide/sugar-blog-tutorial/01-项目初始化' },
        // ... 更多文章
      ]
    }
  ],
  
  // /sop/ 路径下的页面使用这个侧边栏
  '/sop/': [
    {
      text: 'SOP',
      items: [
        { text: '快速开始', link: '/sop/component.md' },
        { text: '配置', link: '/sop/more' },
      ]
    }
  ],
}
```

### 侧边栏分组技巧

- 按**教程系列**分组：每套系列作为一个分组
- 按**功能模块**分组：配置、文章、部署等
- 可以为不同的路径前缀配置不同的侧边栏

## 社交链接

在导航栏右侧显示社交图标：

```typescript
socialLinks: [
  {
    icon: 'github',
    link: 'https://github.com/yangshuyin98/my-tech-blog.git'
  }
]
```

支持 `github`、`twitter`、`discord` 等内置图标。

## 其他导航配置

```typescript
themeConfig: {
  // 目录深度
  outline: {
    level: [2, 3],      // 显示 h2、h3
    label: '目录'
  },
  
  // 文案定制
  returnToTopLabel: '回到顶部',
  sidebarMenuLabel: '相关文章',
  lastUpdatedText: '上次更新于',
  
  // 网站 Logo
  logo: '/logo.png',
}
```

## 完整示例

本博客的配置（`config.mts`）：

```typescript
export default defineConfig({
  extends: blogTheme,
  base: process.env.GITHUB_ACTIONS === 'true' ? '/my-tech-blog' : '/',
  lang: 'zh-cn',
  title: '我的博客',
  description: 'shuyin的博客',
  
  themeConfig: {
    outline: { level: [2, 3], label: '目录' },
    returnToTopLabel: '回到顶部',
    sidebarMenuLabel: '相关文章',
    lastUpdatedText: '上次更新于',
    logo: '/logo.png',
    
    nav: [ /* 见上文 */ ],
    sidebar: { /* 见上文 */ },
    socialLinks: [{ icon: 'github', link: 'https://github.com/yangshuyin98/my-tech-blog.git' }],
  }
})
```
